Store Discovery

                    
GET

                    
/stores/{zipcode}

                

                
                

                    Discover grocery stores in a specific ZIP code. Optionally filter by store chain.
                


                

                    
Parameters

                    

                        

                            
zipcode

                            
string

                            
required

                            
5-digit ZIP code

                        

                        

                            
store_chain

                            
string

                            
optional

                            
Filter by store chain (e.g., "Target", "Walmart")

                        

                    

                


                

                    
Try it

                    

                        

                            
                            
                        

                        

                            
                            
                        

                        
                    

                    
                    

                        

                            Status: -
                            Time: -ms
                            
                        

                        

                    

                


                

                    

                        
                        
                        
                    

                    
                    

                        

                            
import requests

# Find stores in NYC
response = requests.get('http://localhost:8000/stores/10001')
print(response.json())

# Find Target stores in NYC
response = requests.get('http://localhost:8000/stores/10001?store_chain=Target')
print(response.json())

                        

                        

                            
// Find stores in NYC
const response = await fetch('http://localhost:8000/stores/10001');
const data = await response.json();
console.log(data);

// Find Target stores in NYC
const targetResponse = await fetch('http://localhost:8000/stores/10001?store_chain=Target');
const targetData = await targetResponse.json();
console.log(targetData);

                        

                        

                            
# Find stores in NYC
curl -X GET "http://localhost:8000/stores/10001"

# Find Target stores in NYC
curl -X GET "http://localhost:8000/stores/10001?store_chain=Target"

                        

                    

                

            


            
            

                

                    
Product Search

                    
GET

                    
/products/search

                

                
                

                    Search for grocery products using AI-powered structured data extraction. Returns detailed product information including price, quantity, images, and store location.
                


                

                    
Parameters

                    

                        

                            
query

                            
string

                            
required

                            
Product search query

                        

                        

                            
store_name

                            
string

                            
optional

                            
Filter by specific store

                        

                        

                            
zipcode

                            
string

                            
optional

                            
5-digit ZIP code for location

                        

                        

                            
context

                            
string

                            
optional

                            
Search context: price_comparison, availability, nutritional, brand_comparison, seasonal

                        

                        

                            
refresh

                            
boolean

                            
optional

                            
Bypass cache for fresh results

                        

                    

                


                

                    
Try it

                    

                        

                            
                            
                        

                        

                            
                            
                        

                        

                            
                            
                        

                        

                            
                            
                        

                        

                            
                        

                        
                    

                    
                    

                        

                            Status: -
                            Time: -ms
                            
                        

                        

                    

                


                

                    

                        
                        
                        
                    

                    
                    

                        

                            
import requests

# Basic product search
response = requests.get('http://localhost:8000/products/search', params={
    'query': 'organic milk',
    'zipcode': '10001',
    'refresh': True
})
print(response.json())

# Price comparison context
response = requests.get('http://localhost:8000/products/search', params={
    'query': 'cheap milk',
    'zipcode': '10001',
    'context': 'price_comparison',
    'refresh': True
})
print(response.json())

                        

                        

                            
// Basic product search
const response = await fetch('http://localhost:8000/products/search?query=organic%20milk&zipcode=10001&refresh=true');
const data = await response.json();
console.log(data);

// Price comparison context
const priceResponse = await fetch('http://localhost:8000/products/search?query=cheap%20milk&zipcode=10001&context=price_comparison&refresh=true');
const priceData = await priceResponse.json();
console.log(priceData);

                        

                        

                            
# Basic product search
curl -X GET "http://localhost:8000/products/search?query=organic%20milk&zipcode=10001&refresh=true"

# Price comparison context
curl -X GET "http://localhost:8000/products/search?query=cheap%20milk&zipcode=10001&context=price_comparison&refresh=true"

                        

                    

                

            


            
            

                

                    
Aggregate Search

                    
GET

                    
/products/aggregate

                

                
                

                    Search for products across multiple stores in a ZIP code and aggregate the results. Perfect for price comparison and finding the best deals.
                


                

                    
Parameters

                    

                        

                            
query

                            
string

                            
required

                            
Product search query

                        

                        

                            
zipcode

                            
string

                            
required

                            
5-digit ZIP code for location

                        

                        

                            
stores

                            
string

                            
optional

                            
Comma-separated store names (e.g., "target,walmart,whole_foods")

                        

                        

                            
radius_miles

                            
integer

                            
optional

                            
Search radius in miles (default: 10)

                        

                        

                            
refresh

                            
boolean

                            
optional

                            
Bypass cache for fresh results

                        

                    

                


                

                    
Try it

                    

                        

                            
                            
                        

                        

                            
                            
                        

                        

                            
                            
                        

                        

                            
                            
                        

                        

                            
                        

                        
                    

                    
                    

                        

                            Status: -
                            Time: -ms
                            
                        

                        

                    

                


                

                    

                        
                        
                        
                    

                    
                    

                        

                            
import requests

# Multi-store search
response = requests.get('http://localhost:8000/products/aggregate', params={
    'query': 'organic milk',
    'zipcode': '10001',
    'stores': 'target,walmart,whole_foods',
    'refresh': True
})
print(response.json())

# Price comparison across all stores
response = requests.get('http://localhost:8000/products/aggregate', params={
    'query': 'cheap bread',
    'zipcode': '10001',
    'radius_miles': 5,
    'refresh': True
})
print(response.json())

                        

                        

                            
// Multi-store search
const response = await fetch('http://localhost:8000/products/aggregate?query=organic%20milk&zipcode=10001&stores=target,walmart,whole_foods&refresh=true');
const data = await response.json();
console.log(data);

// Price comparison across all stores
const priceResponse = await fetch('http://localhost:8000/products/aggregate?query=cheap%20bread&zipcode=10001&radius_miles=5&refresh=true');
const priceData = await priceResponse.json();
console.log(priceData);

                        

                        

                            
# Multi-store search
curl -X GET "http://localhost:8000/products/aggregate?query=organic%20milk&zipcode=10001&stores=target,walmart,whole_foods&refresh=true"

# Price comparison across all stores
curl -X GET "http://localhost:8000/products/aggregate?query=cheap%20bread&zipcode=10001&radius_miles=5&refresh=true"
